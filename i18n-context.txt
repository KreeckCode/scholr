from django.conf import settings
from django.http import Http404


class FeatureFlagMiddleware:
    """
    Blocks URL prefixes for disabled product modules.
    Only active when the corresponding flag is False.
    """

    # The shop is mounted at /courses/ in scholr/urls.py
    LMS_PREFIXES = ('/courses/',)
    ACADEMICS_PREFIXES = ('/academics/',)
    PARTNERS_PREFIXES = ('/partners/',)

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        path = request.path

        if not getattr(settings, 'SCHOLR_ACADEMICS_ENABLED', True):
            if path.startswith(self.ACADEMICS_PREFIXES):
                raise Http404

        if not getattr(settings, 'SCHOLR_LMS_ENABLED', True):
            if path.startswith(self.LMS_PREFIXES):
                raise Http404

        if not getattr(settings, 'SCHOLR_PARTNERS_ENABLED', False):
            if path.startswith(self.PARTNERS_PREFIXES):
                raise Http404

        return self.get_response(request)

class User(AbstractUser):
    """
    Custom user model for scholr CAS
    Extends Django's AbstractUser to support role-based access and African context needs
    Phone numbers are primary contact method since many pastors lack email
    """
    
    # Role choices for the system
    STUDENT = 'student'
    DEAN = 'dean'
    ADMIN = 'admin'

    # School management roles - academics app context
    TEACHER = 'teacher'
    PRINCIPAL = 'principal'
    HOD = 'hod'
    ADMIN_CLERK = 'admin_clerk'
    PARENT = 'parent'
    LEARNER = 'learner'

    # Partner network roles - partners app context
    PROGRAMME_STAFF = 'programme_staff'
    PARTNER = 'partner'

    ROLE_CHOICES = [
        # LMS roles
        (STUDENT, 'Student (Pastor)'),
        (DEAN, 'Dean (Facilitator)'),
        (ADMIN, 'System Administrator'),
        # School roles - added in academics Stage 1
        (TEACHER, 'Teacher'),
        (PRINCIPAL, 'Principal'),
        (HOD, 'Head of Department'),
        (ADMIN_CLERK, 'Administrative Clerk'),
        (PARENT, 'Parent / Guardian'),
        (LEARNER, 'Learner'),
        # Partner network roles
        (PROGRAMME_STAFF, 'Programme Staff'),
        (PARTNER, 'Partner Staff'),
    ]
    # Role groupings - I'm using these to avoid scattering role checks everywhere
    LMS_ROLES = {STUDENT, DEAN, ADMIN}
    SCHOOL_ROLES = {TEACHER, PRINCIPAL, HOD, ADMIN_CLERK, PARENT, LEARNER}
    STAFF_ROLES = {TEACHER, PRINCIPAL, HOD, ADMIN_CLERK}
    PARTNER_NETWORK_ROLES = {PROGRAMME_STAFF, PARTNER}
    
    # Phone validator for international format
    phone_regex = RegexValidator(
        regex=r'^\+?1?\d{9,15}$',
        message="Phone must be in format: '+999999999'. Up to 15 digits allowed."
    )
    
    # Core fields
    role = models.CharField(
        max_length=20,
        choices=ROLE_CHOICES,
        default=STUDENT,
        db_index=True,
        help_text="User's role in the system determines permissions"
    )
    
    phone_number = models.CharField(
        max_length=20,
        unique=True,
        null=True,
        blank=True,
    )
    
    # Email is optional in African rural context
    email = models.EmailField(
        _('email address'),
        blank=True,
        null=True,
        help_text="Email is optional - many pastors don't have regular access"
    )
    
    # Account status fields
    is_verified = models.BooleanField(
        default=False,
        help_text="Whether dean/admin has verified this account"
    )
    
    verified_by = models.ForeignKey(
        'self',
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name='verified_users',
        help_text="Admin/Dean who verified this account"
    )
    
    verified_at = models.DateTimeField(
        null=True,
        blank=True,
        help_text="When the account was verified"
    )
    
    # Tracking fields
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    last_sync = models.DateTimeField(
        null=True,
        blank=True,
        help_text="Last time this user's data was synced from offline device"
    )
    
    class Meta:
        db_table = 'users'
        verbose_name = 'User'
        verbose_name_plural = 'Users'
        ordering = ['-created_at']
        indexes = [
            models.Index(fields=['email'], name='idx_user_email'),
            models.Index(fields=['phone_number'], name='idx_user_phone'),
            models.Index(fields=['role', 'is_active'], name='idx_user_role_active'),
            models.Index(fields=['is_verified'], name='idx_user_verified'),
            models.Index(fields=['created_at'], name='idx_user_created'),
        ]
    
    def __str__(self):
        return f"{self.get_full_name() or self.username} ({self.get_role_display()})"
    
    def get_full_name(self):
        """
        Return full name or username if name not set
        Handles cases where pastors may only provide one name
        """
        full_name = f"{self.first_name} {self.last_name}".strip()
        return full_name if full_name else self.username
    
    def can_verify_users(self):
        """Check if user can verify other accounts - LMS deans/admins and school principals."""
        return self.role in {self.DEAN, self.ADMIN, self.PRINCIPAL} and self.is_active

    def can_manage_content(self):
        """Check if user can upload/manage course content - LMS admins only."""
        return self.role == self.ADMIN and self.is_active

    def is_school_staff(self):
        """True for any role that works inside the school building."""
        return self.role in self.STAFF_ROLES and self.is_active

    def is_lms_user(self):
        """True for pastoral LMS roles - determines Moodle sync eligibility."""
        return self.role in self.LMS_ROLES

    def get_students(self):
        """
        For deans: get LMS students in their region.
        For admins: get all LMS students.
        School learners are fetched via academics.Learner, not here.
        """
        if self.role == self.DEAN:
            return User.objects.filter(
                role=self.STUDENT,
                profile__region=self.profile.region
            ).select_related('profile')
        elif self.role == self.ADMIN:
            return User.objects.filter(role=self.STUDENT).select_related('profile')
        return User.objects.none()

class Order(models.Model):
    """
    Customer orders - tracks purchases and payments
    FREE products create orders with total_amount = 0 and auto-complete
    """
    
    # Order status choices
    PENDING = 'pending'
    PROCESSING = 'processing'
    COMPLETED = 'completed'
    FAILED = 'failed'
    REFUNDED = 'refunded'
    FREE = 'free'  # For free course "orders"
    
    STATUS_CHOICES = [
        (PENDING, 'Pending Payment'),
        (PROCESSING, 'Processing Payment'),
        (COMPLETED, 'Completed'),
        (FAILED, 'Payment Failed'),
        (REFUNDED, 'Refunded'),
        (FREE, 'Free Enrollment'),
    ]
    
    # Core fields
    order_number = models.CharField(
        max_length=50,
        unique=True,
        db_index=True,
        help_text="Unique order identifier"
    )
    user = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name='orders'
    )
    
    # Pricing
    subtotal = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        default=Decimal('0.00')
    )
    discount_amount = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        default=Decimal('0.00')
    )
    total_amount = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        default=Decimal('0.00')
    )
    currency = models.CharField(max_length=3, default='ZAR')
    
    # Status
    status = models.CharField(
        max_length=20,
        choices=STATUS_CHOICES,
        default=PENDING,
        db_index=True
    )
    
    # Payment details
    payment_method = models.CharField(
        max_length=50,
        blank=True,
        help_text="Payment gateway used (paystack, stripe, free)"
    )
    payment_reference = models.CharField(
        max_length=200,
        blank=True,
        db_index=True,
        help_text="Payment gateway transaction reference"
    )
    payment_metadata = models.JSONField(
        default=dict,
        blank=True,
        help_text="Additional payment data from gateway"
    )
    
    # Timestamps
    paid_at = models.DateTimeField(null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    # Enrollment tracking
    enrollment_processed = models.BooleanField(
        default=False,
        db_index=True,
        help_text="Whether enrollments have been created"
    )
    enrollment_processed_at = models.DateTimeField(null=True, blank=True)
    enrollment_error = models.TextField(
        blank=True,
        help_text="Error message if enrollment failed"
    )
    # Proof of payment (manual EFT flow)
    proof_of_payment = models.FileField(
        upload_to='proofs_of_payment/%Y/%m/',
        blank=True,
        null=True,
        help_text="Customer uploaded proof of payment"
    )
    pop_submitted_at = models.DateTimeField(
        null=True,
        blank=True,
        help_text="When customer submitted proof of payment"
    )
    
    # Notes
    customer_notes = models.TextField(blank=True)
    admin_notes = models.TextField(blank=True)
    
    class Meta:
        db_table = 'orders'
        verbose_name = 'Order'
        verbose_name_plural = 'Orders'
        ordering = ['-created_at']
        indexes = [
            models.Index(fields=['user', 'status'], name='idx_order_user_status'),
            models.Index(fields=['order_number'], name='idx_order_number'),
            models.Index(fields=['status', 'created_at'], name='idx_order_status_created'),
            models.Index(fields=['payment_reference'], name='idx_order_payment_ref'),
            models.Index(fields=['created_at'], name='idx_order_created'),
            models.Index(fields=['paid_at'], name='idx_order_paid'),
        ]
    
    def __str__(self):
        return f"Order {self.order_number} - {self.user.get_full_name()} (R{self.total_amount})"
    
    def save(self, *args, **kwargs):
        """Generate order number if not set"""
        if not self.order_number:
            import uuid
            self.order_number = f"JH{timezone.now().strftime('%Y%m%d')}{uuid.uuid4().hex[:8].upper()}"
        super().save(*args, **kwargs)
    
    def is_paid(self):
        """Check if order is paid"""
        return self.status in [self.COMPLETED, self.FREE]
    
    def is_free_order(self):
        """Check if this is a free order"""
        return self.total_amount == Decimal('0.00') or self.status == self.FREE
    
    def calculate_totals(self):
        """Recalculate order totals from items"""
        items = self.items.all()
        self.subtotal = sum(item.subtotal for item in items)
        self.total_amount = self.subtotal - self.discount_amount
        
        # Free order if total is 0
        if self.total_amount == Decimal('0.00'):
            self.status = self.FREE
            self.payment_method = 'free'
    
    def mark_paid(self, payment_reference='', payment_method=''):
        """Mark order as paid and trigger enrollment"""
        self.status = self.COMPLETED
        self.paid_at = timezone.now()
        if payment_reference:
            self.payment_reference = payment_reference
        if payment_method:
            self.payment_method = payment_method
        self.save()
    
    def process_enrollment(self):
        """
        Trigger enrollment processing
        Called after payment confirmation or immediately for free orders
        """
        from .services import EnrollmentService
        
        if self.enrollment_processed:
            return True
        
        try:
            service = EnrollmentService()
            enrollments = service.process_order_enrollment(self)
            
            self.enrollment_processed = True
            self.enrollment_processed_at = timezone.now()
            self.save()
            
            return True
        except Exception as e:
            self.enrollment_error = str(e)
            self.save()
            return False


class OrderItem(models.Model):
    """
    Individual items in an order
    One order can have multiple products
    """
    order = models.ForeignKey(
        'Order',
        on_delete=models.CASCADE,
        related_name='items'
    )
    product = models.ForeignKey(
        'Product',
        on_delete=models.PROTECT,
        related_name='order_items'
    )
    
    # Pricing at time of purchase (products can change price later)
    quantity = models.PositiveIntegerField(default=1)
    unit_price = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        help_text="Price per unit at time of purchase"
    )
    subtotal = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        help_text="quantity × unit_price"
    )
    
    # Track what was enrolled from this item
    enrollment_log = models.JSONField(
        default=dict,
        blank=True,
        help_text="Courses enrolled from this product"
    )
    
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        db_table = 'order_items'
        verbose_name = 'Order Item'
        verbose_name_plural = 'Order Items'
        ordering = ['id']
    
    def __str__(self):
        return f"{self.product.name} × {self.quantity}"
    
    def save(self, *args, **kwargs):
        """Calculate subtotal automatically"""
        self.subtotal = self.unit_price * self.quantity
        super().save(*args, **kwargs)

class MoodleIntegrationService:
    """
    Service layer for Moodle integration
    Handles all sync operations and business logic
    """
    
    def __init__(self, moodle_instance: MoodleInstance = None):
        """
        Initialize service with a Moodle instance
        
        Args:
            moodle_instance: MoodleInstance to use (uses primary if None)
        """
        if moodle_instance is None:
            moodle_instance = MoodleInstance.objects.filter(
                is_active=True,
                is_primary=True
            ).first()
            
            if not moodle_instance:
                raise ValueError("No active Moodle instance configured")
        
        self.moodle_instance = moodle_instance
        self.client = MoodleClient(
            base_url=moodle_instance.base_url,
            token=moodle_instance.ws_token
        )
    
    def test_connection(self) -> Dict:
        """
        Test connection to Moodle and update health status
        
        Returns:
            Dict with connection test results
        """
        try:
            result = self.client.test_connection()
            
            if result['success']:
                self.moodle_instance.mark_healthy()
                logger.info(f"Moodle connection healthy: {result['site_name']}")
            else:
                self.moodle_instance.mark_offline(result.get('error', 'Unknown error'))
                logger.error(f"Moodle connection failed: {result.get('error')}")
            
            return result
            
        except Exception as e:
            error_msg = str(e)
            self.moodle_instance.mark_offline(error_msg)
            logger.exception("Moodle connection test failed")
            return {
                'success': False,
                'error': error_msg
            }
    
    # ==================== CATEGORY SYNC ====================
    
    def sync_categories(self, initiated_by: User = None) -> SyncLog:
        """
        Sync course categories to Moodle
        Creates categories that don't exist, updates existing ones
        
        Args:
            initiated_by: User who triggered the sync
            
        Returns:
            SyncLog instance with results
        """
        sync_log = SyncLog.objects.create(
            sync_type='categories',
            initiated_by=initiated_by,
            moodle_instance=self.moodle_instance
        )
        
        try:
            sync_log.mark_running()
            
            categories = CourseCategory.objects.filter(is_active=True)
            sync_log.total_items = categories.count()
            sync_log.save(update_fields=['total_items'])
            
            # Get existing Moodle categories
            moodle_categories = {cat['name']: cat for cat in self.client.get_categories()}
            
            results = {
                'created': [],
                'updated': [],
                'skipped': [],
                'errors': []
            }
            
            for category in categories:
                try:
                    moodle_cat = moodle_categories.get(category.name)
                    
                    if moodle_cat:
                        # Category exists, update if needed
                        category.moodle_category_id = moodle_cat['id']
                        category.last_synced = timezone.now()
                        category.save(update_fields=['moodle_category_id', 'last_synced'])
                        
                        results['skipped'].append(category.name)
                        sync_log.skipped_items += 1
                    else:
                        # Create new category
                        moodle_id = self.client.create_category(
                            name=category.name,
                            description=category.description
                        )
                        
                        category.moodle_category_id = moodle_id
                        category.last_synced = timezone.now()
                        category.save(update_fields=['moodle_category_id', 'last_synced'])
                        
                        results['created'].append(category.name)
                        sync_log.successful_items += 1
                    
                except Exception as e:
                    error_msg = f"Failed to sync category {category.name}: {str(e)}"
                    results['errors'].append(error_msg)
                    sync_log.failed_items += 1
                    logger.error(error_msg)
            
            sync_log.details = results
            sync_log.mark_completed()
            
            logger.info(f"Category sync completed: {sync_log.successful_items} successful, {sync_log.failed_items} failed")
            return sync_log
            
        except Exception as e:
            error_msg = f"Category sync failed: {str(e)}"
            sync_log.mark_failed(error_msg)
            logger.exception(error_msg)
            return sync_log
    
    # ==================== USER SYNC ====================
    
    def sync_users(self, initiated_by: User = None, user_ids: List[int] = None) -> SyncLog:
        """
        Sync verified users to Moodle
        Only syncs users who are verified and don't have Moodle accounts
        
        Args:
            initiated_by: User who triggered the sync
            user_ids: Specific user IDs to sync (syncs all verified if None)
            
        Returns:
            SyncLog instance with results
        """
        sync_log = SyncLog.objects.create(
            sync_type='users',
            initiated_by=initiated_by,
            moodle_instance=self.moodle_instance
        )
        
        try:
            sync_log.mark_running()
            
            # Get users to sync - only verified students
            users_query = User.objects.filter(
                role=User.STUDENT,
                is_verified=True,
                is_active=True
            )
            
            if user_ids:
                users_query = users_query.filter(id__in=user_ids)
            
            users = users_query.select_related('profile')
            sync_log.total_items = users.count()
            sync_log.save(update_fields=['total_items'])
            
            results = {
                'created': [],
                'updated': [],
                'skipped': [],
                'errors': []
            }
            
            for user in users:
                try:
                    # Check if user already has mapping
                    mapping, created = MoodleUserMapping.objects.get_or_create(
                        cas_user=user,
                        defaults={
                            'moodle_username': user.username,
                            'sync_status': 'pending'
                        }
                    )
                    
                    # If user already synced, skip
                    if mapping.sync_status == 'synced' and mapping.moodle_user_id:
                        results['skipped'].append(user.username)
                        sync_log.skipped_items += 1
                        continue
                    
                    # Check if user exists in Moodle
                    existing_user = self.client.get_user_by_username(user.username)
                    
                    if existing_user:
                        # User exists, just map it
                        mapping.moodle_user_id = existing_user['id']
                        mapping.mark_synced()
                        
                        results['skipped'].append(user.username)
                        sync_log.skipped_items += 1
                    else:
                        # Create new user in Moodle
                        user_data = {
                            'username': user.username,
                            'password': generate_moodle_password(),
                            'firstname': user.first_name,
                            'lastname': user.last_name,
                            'email': user.email or f"{user.username}@judea.local",
                            'phone1': user.phone_number,
                            'city': user.profile.region if hasattr(user, 'profile') else '',
                            'country': self._get_country_code(user.profile.country) if hasattr(user, 'profile') else 'ZA',
                        }
                        
                        moodle_user_id = self.client.create_user(user_data)
                        
                        mapping.moodle_user_id = moodle_user_id
                        mapping.mark_synced()
                        
                        results['created'].append(user.username)
                        sync_log.successful_items += 1
                        
                        # Auto-enroll if enabled
                        self._auto_enroll_user(user, moodle_user_id)
                    
                except Exception as e:
                    error_msg = f"Failed to sync user {user.username}: {str(e)}"
                    results['errors'].append(error_msg)
                    
                    if hasattr(user, 'moodle_mapping'):
                        user.moodle_mapping.mark_failed(error_msg)
                    
                    sync_log.failed_items += 1
                    logger.error(error_msg)
            
            sync_log.details = results
            sync_log.mark_completed()
            
            logger.info(f"User sync completed: {sync_log.successful_items} successful, {sync_log.failed_items} failed")
            return sync_log
            
        except Exception as e:
            error_msg = f"User sync failed: {str(e)}"
            sync_log.mark_failed(error_msg)
            logger.exception(error_msg)
            return sync_log
    
    def _auto_enroll_user(self, user: User, moodle_user_id: int):
        """
        Auto-enroll user in courses with auto_enroll_enabled
        
        Args:
            user: CAS user
            moodle_user_id: Moodle user ID
        """
        auto_enroll_courses = Course.objects.filter(
            auto_enroll_enabled=True,
            visible=True,
            sync_status='synced',
            moodle_course_id__isnull=False
        )
        
        for course in auto_enroll_courses:
            try:
                # Check if already enrolled
                enrollment, created = CourseEnrollment.objects.get_or_create(
                    user=user,
                    course=course,
                    defaults={
                        'is_active': True,
                        'sync_status': 'pending'
                    }
                )
                
                if created:
                    # Enroll in Moodle
                    self.client.enrol_user(moodle_user_id, course.moodle_course_id)
                    enrollment.mark_synced()
                    logger.info(f"Auto-enrolled {user.username} in {course.short_name}")
                    
            except Exception as e:
                logger.error(f"Failed to auto-enroll {user.username} in {course.short_name}: {str(e)}")
    
    # ==================== COURSE SYNC ====================
    
    def sync_courses(self, initiated_by: User = None, course_ids: List[int] = None) -> SyncLog:
        """
        Sync courses to Moodle with smart duplicate handling
        
        Flow:
        1. If course has moodle_course_id → verify it exists
        2. If not found by ID → search by shortname
        3. If found → update it and reconcile
        4. If not found → create new
        5. Handle duplicates gracefully
        """
        sync_log = SyncLog.objects.create(
            sync_type='courses',
            initiated_by=initiated_by,
            moodle_instance=self.moodle_instance
        )
        
        try:
            sync_log.mark_running()

    def sync_enrollments(self, initiated_by: User = None) -> SyncLog:
        """
        Sync enrollments to Moodle
        Enrolls/unenrolls students based on CourseEnrollment records
        
        Args:
            initiated_by: User who triggered the sync
            
        Returns:
            SyncLog instance with results
        """

            sync_type='enrollments',
            initiated_by=initiated_by,
            moodle_instance=self.moodle_instance
        )
        
        try:

                        )
                        enrollment.mark_synced()
                        
                        results['enrolled'].append(
                            f"{enrollment.user.username} → {enrollment.course.short_name}"
                        )
                        sync_log.successful_items += 1
                    else:
                        # Unenroll user
                        self.client.unenrol_user(
                            mapping.moodle_user_id,
                            enrollment.course.moodle_course_id
                        )
                        enrollment.mark_synced()
                        
                        results['unenrolled'].append(
                            f"{enrollment.user.username} ← {enrollment.course.short_name}"
                        )
                        sync_log.successful_items += 1
                    
                except Exception as e:
                    error_msg = f"Failed to sync enrollment {enrollment}: {str(e)}"
                    results['errors'].append(error_msg)
                    enrollment.mark_failed(error_msg)
                    sync_log.failed_items += 1
                    logger.error(error_msg)
            
            sync_log.details = results
            sync_log.mark_completed()
            
            logger.info(f"Enrollment sync completed: {sync_log.successful_items} successful, {sync_log.failed_items} failed")

    def full_sync(self, initiated_by: User = None) -> SyncLog:
        """
        Perform a full sync of all entities
        Order: Categories → Courses → Users → Enrollments
        
        Args:
            initiated_by: User who triggered the sync
            
        Returns:
            SyncLog instance with results
        """
        sync_log = SyncLog.objects.create(
            sync_type='full',
            initiated_by=initiated_by,
            moodle_instance=self.moodle_instance
        )
        
        try:
            sync_log.mark_running()
            
            results = {
                'categories': None,
                'courses': None,
                'users': None,
                'enrollments': None
            }
            
            # Step 1: Sync categories
            cat_log = self.sync_categories(initiated_by)
            results['categories'] = {
                'successful': cat_log.successful_items,
                'failed': cat_log.failed_items
            }
            
            # Step 2: Sync courses
            course_log = self.sync_courses(initiated_by)
            results['courses'] = {
                'successful': course_log.successful_items,
                'failed': course_log.failed_items
            }
            
            # Step 3: Sync users
            user_log = self.sync_users(initiated_by)
            results['users'] = {
                'successful': user_log.successful_items,
                'failed': user_log.failed_items
            }
            
            # Step 4: Sync enrollments
            enrol_log = self.sync_enrollments(initiated_by)
            results['enrollments'] = {
                'successful': enrol_log.successful_items,
                'failed': enrol_log.failed_items
            }
            
            # Calculate totals
            sync_log.total_items = sum([
                cat_log.total_items,
                course_log.total_items,
                user_log.total_items,
                enrol_log.total_items
            ])
            
            sync_log.successful_items = sum([
                cat_log.successful_items,
                course_log.successful_items,
                user_log.successful_items,
                enrol_log.successful_items
            ])
            
            sync_log.failed_items = sum([
                cat_log.failed_items,
                course_log.failed_items,
                user_log.failed_items,
                enrol_log.failed_items
            ])
            
            sync_log.details = results
            sync_log.mark_completed()
            
            logger.info(f"Full sync completed: {sync_log.successful_items}/{sync_log.total_items} successful")
            return sync_log

            sync_log.mark_failed(error_msg)

@shared_task(bind=True, max_retries=3)
def process_order_enrollment_task(self, order_id):
    """
    Background task to process order enrollment
    Prevents blocking HTTP requests during enrollment
    """
    from accounts.models import Order
    from shop.services import EnrollmentService
    
    try:
        order = Order.objects.select_related('user').get(id=order_id)
        
        if not order.is_paid():
            logger.warning(f"Order {order.order_number} not paid, skipping enrollment")
            return
        
        if order.enrollment_processed:
            logger.warning(f"Order {order.order_number} already processed")
            return
        
        service = EnrollmentService()
        enrollments = service.process_order_enrollment(order)
        
        # Clear user's course cache
        cache.delete(f'user_courses:{order.user.id}')
        cache.delete(f'user_enrollments:{order.user.id}')
        
        logger.info(f"Successfully processed enrollment for order {order.order_number}: {len(enrollments)} products")
        
        return {
            'order_number': order.order_number,
            'enrollments_created': len(enrollments),
            'status': 'success'
        }
        
    except Order.DoesNotExist:
        logger.error(f"Order {order_id} not found")
        raise
    
    except Exception as exc:
        logger.error(f"Error processing enrollment for order {order_id}: {str(exc)}")
        raise self.retry(exc=exc, countdown=60 * (self.request.retries + 1))

"""
Celery tasks for the academics module.

Beat schedule (configured in settings.CELERY_BEAT_SCHEDULE):
  04:00 — sync_online_learning_progress_task   pull Moodle progress
  06:00 — send_stall_notifications_task        email stall alerts

Bulk tasks fire on demand from the bulk operation views:
  bulk_create_lms_accounts_task
  bulk_enroll_in_course_task
"""
import logging

from celery import shared_task

logger = logging.getLogger(__name__)


# ==================== NIGHTLY SYNC ====================

@shared_task(bind=True, max_retries=3, default_retry_delay=300)
def sync_online_learning_progress_task(self):
    """
    Pull Moodle progress for every learner with an active online enrollment.
    Runs at 04:00 via Beat, before the stall notification task.

    After all learners are synced, auto-marks any enrollment that reached
    100% as STATUS_COMPLETED — a single bulk UPDATE rather than per-row logic.
    """
    from scholr.features import bridge_enabled

    if not bridge_enabled():
        logger.info('sync_online_learning_progress_task: bridge disabled, skipping.')
        return {'skipped': True}

    from academics.models.lms_link import OnlineEnrollment
    from academics.models.learner import Learner
    from academics.services.lms_bridge import LMSBridgeService

    learner_ids = list(
        OnlineEnrollment.objects.filter(
            status=OnlineEnrollment.STATUS_ACTIVE,
        ).values_list('learner_id', flat=True).distinct()
    )

    service = LMSBridgeService()
    results = {
        'total_learners': len(learner_ids),
        'synced':         0,
        'failed':         0,
        'auto_completed': 0,
    }

    for learner_id in learner_ids:
        try:
            learner = Learner.objects.get(pk=learner_id)
            result  = service.sync_progress_for_learner(learner)
            if result.get('success'):
                results['synced'] += 1
            else:
                results['failed'] += 1
                logger.warning(
                    f'Progress sync returned non-success for learner '
                    f'{learner_id}: {result.get("error")}'
                )
        except Learner.DoesNotExist:
            logger.error(f'Progress sync: learner {learner_id} not found.')
            results['failed'] += 1
        except Exception as exc:
            logger.error(f'Progress sync failed for learner {learner_id}: {exc}')
            results['failed'] += 1
            try:
                raise self.retry(exc=exc)
            except self.MaxRetriesExceededError:
                pass

    # Auto-completion: any active enrollment sitting at 100% is complete.
    # Single bulk UPDATE — no per-row overhead.
    completed = OnlineEnrollment.objects.filter(
        status=OnlineEnrollment.STATUS_ACTIVE,
        progress_percentage__gte=100,
    ).update(status=OnlineEnrollment.STATUS_COMPLETED)
    results['auto_completed'] = completed

    if completed:
        logger.info(f'Auto-completed {completed} online enrollment(s).')

    logger.info(
        f'sync_online_learning_progress_task complete: '
        f'{results["synced"]} synced, {results["failed"]} failed, '
        f'{results["auto_completed"]} auto-completed '
        f'of {results["total_learners"]} learners.'
    )
    return results

@shared_task(bind=True, max_retries=2, default_retry_delay=600)
def send_stall_notifications_task(self):
    """
    Detect stalled online course progress and email relevant staff.
    Runs at 06:00 via Beat — 2 hours after the sync task so data is fresh.

    Thresholds (override via .env):
      PROGRESS_STALL_DAYS=7
      PROGRESS_NOT_STARTED_DAYS=3
      PROGRESS_ALERT_COOLDOWN_DAYS=7
    """
    from scholr.features import bridge_enabled

    if not bridge_enabled():
        logger.info('send_stall_notifications_task: bridge disabled, skipping.')
        return {'skipped': True}

    try:
        from academics.emails import ProgressStallEmailService
        result = ProgressStallEmailService().detect_and_send()
    except Exception as exc:
        logger.error(f'send_stall_notifications_task failed: {exc}')
        try:
            raise self.retry(exc=exc)
        except self.MaxRetriesExceededError:
            return {'error': str(exc)}

    logger.info(
        f'send_stall_notifications_task complete: '
        f'{result["stalled"]} stalled, '
        f'{result["not_started"]} not started, '
        f'{result["emails_sent"]} emails sent, '
        f'{result["skipped_no_recipients"]} skipped (no recipients).'
    )
    return result

        f'{results["created"]} created, '
        f'{results["already_existed"]} already existed, '
        f'{results["failed"]} failed '
        f'of {results["total"]} learners.'
    )
    return results


@shared_task(bind=True, max_retries=2, default_retry_delay=120)
def bulk_enroll_in_course_task(
    self, learner_ids, course_id, enrolled_by_id, academic_year_id,
):
    """
    Enrol a batch of learners in a Moodle course.

    Fired by BulkEnrollView. Per-learner skips:
      - No LMS account or account not synced to Moodle
      - Already enrolled (enroll_in_course raises ValueError)

    The task never aborts on a single failure — it logs and continues
    so one bad record doesn't block the rest of the cohort.
    """
    from scholr.features import bridge_enabled

    if not bridge_enabled():
        return {'skipped': True}

    from accounts.models import User
    from academics.models.learner import Learner
    from academics.models.lms_link import LearnerLMSAccount
    from academics.models.school import AcademicYear
    from academics.services.lms_bridge import LMSBridgeService
    from integration.models import Course as LMSCourse

    # Resolve all FK objects up-front; fail fast if any are missing
    try:
        enrolled_by   = User.objects.get(pk=enrolled_by_id)
        course        = LMSCourse.objects.get(pk=course_id)
        academic_year = AcademicYear.objects.get(pk=academic_year_id)
    except Exception as exc:
        logger.error(f'bulk_enroll_in_course_task: setup failed: {exc}')
        return {'error': str(exc)}

    service = LMSBridgeService()
    results = {
        'total':             len(learner_ids),
        'enrolled':          0,
        'already_enrolled':  0,
        'skipped_no_account': 0,
        'failed':            0,
    }

    for learner_id in learner_ids:
        try:
            learner = Learner.objects.get(pk=learner_id, is_active=True)
        except Learner.DoesNotExist:
            logger.warning(
                f'bulk_enroll_in_course_task: learner {learner_id} not found.'
            )
            results['failed'] += 1
            continue

        # Gate: LMS account must exist and be Moodle-synced
        try:
            lms_account = learner.lms_account
            if not lms_account.is_active or not lms_account.is_synced_to_moodle():
                results['skipped_no_account'] += 1
                continue
        except LearnerLMSAccount.DoesNotExist:
            results['skipped_no_account'] += 1
            continue

        try:
            service.enroll_in_course(
                learner=learner,
                course=course,
                enrolled_by=enrolled_by,
                reason='Bulk enrollment',
                academic_year=academic_year,
            )
            results['enrolled'] += 1
        except ValueError:
            # enroll_in_course raises ValueError when already enrolled
            results['already_enrolled'] += 1
        except Exception as exc:
            logger.error(
                f'bulk_enroll_in_course_task: '
                f'enrolment failed for learner {learner_id}: {exc}'
            )
            results['failed'] += 1

    logger.info(
        f'bulk_enroll_in_course_task [{course.title}]: '
        f'{results["enrolled"]} enrolled, '
        f'{results["already_enrolled"]} already enrolled, '
        f'{results["skipped_no_account"]} no account, '
        f'{results["failed"]} failed '
        f'of {results["total"]} learners.'
    )
    return results===== requirements.txt =====
